-- 1) Create or replace the hardened has_role function
CREATE OR REPLACE FUNCTION public.has_role(p_user_id uuid, p_role text)
RETURNS boolean
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT EXISTS(
    SELECT 1 FROM public.user_roles ur
    WHERE ur.user_id = p_user_id AND ur.role = p_role
  );
$$;

-- 2) Add comment documenting purpose and security assumptions
COMMENT ON FUNCTION public.has_role(uuid, text) IS E'
Return true if the given user_id has the specified role in public.user_roles.

Security notes:
- Declared SECURITY DEFINER to allow checks when caller lacks direct table rights.
- SET search_path = public to prevent search_path injection.
- Function marked STABLE as it does not modify DB state and depends only on table data.
- Revoke EXECUTE from PUBLIC/anon and grant to authenticated and service_role only.
';

-- 3) Privilege hardening
REVOKE EXECUTE ON FUNCTION public.has_role(uuid, text) FROM PUBLIC;
REVOKE EXECUTE ON FUNCTION public.has_role(uuid, text) FROM anon;
GRANT EXECUTE ON FUNCTION public.has_role(uuid, text) TO authenticated;
GRANT EXECUTE ON FUNCTION public.has_role(uuid, text) TO service_role;

-- 4) Create composite index for performance (if user_roles table exists)
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace WHERE c.relname = 'user_roles' AND n.nspname = 'public') THEN
    IF NOT EXISTS (
      SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace

    ) THEN
